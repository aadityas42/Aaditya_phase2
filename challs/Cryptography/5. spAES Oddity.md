# 5. spAES Oddity

## Solution:

- From chal.py and the name of the challenge itself, we come to know that the encryption used is AES-ECB and our input is 'message' which HAS to be an odd number of bytes

- The key is random and unknown.

- We exploited a weakness in AES-ECB, that is, if 2 plaintexts are same, so will the ciphertext.

-So we just make guesses and check if the encrypted guess matches the encrypted flag, 2 bytes at a time.

```
from pwn import *
import string

charset = "nite{}-_ABCD" + string.ascii_lowercase + "01234"
r = remote('spaesoddity.nitephase.live', 45673)

def encrypt(payload_bytes):
    # Convert bytes to hex string for the prompt
    r.sendlineafter(b'input in hex:', payload_bytes.hex().encode())
    # Receive the hex response
    res = r.recvline().strip().decode()
    return bytes.fromhex(res)

def get_block(data, block_idx):
    start = block_idx * 16
    end = start + 16
    return data[start:end]

known_flag = b"nite{"
print(f"[*] Starting attack. Known flag: {known_flag.decode()}")

# Flag is 49 chars long. We solve 2 chars at a time.
# We start from index 5 (after 'nite{')
while len(known_flag) < 49:

    target_len = len(known_flag) + 2
    pad_len = (16 - (target_len % 16)) % 16
    padding = b'A' * pad_len
    
    real_ct = encrypt(padding)
    block_idx = (pad_len + target_len) // 16 - 1
    target_block = get_block(real_ct, block_idx)
    
    found_pair = False
    
    print(f"[*] Cracking bytes {len(known_flag)} and {len(known_flag)+1}...")
    
    for c1 in charset:
        for c2 in charset:
            guess = known_flag + c1.encode() + c2.encode()
            
            # We append b'A' to make length odd
            payload = padding + guess + b'A'
            
            test_ct = encrypt(payload)
            test_block = get_block(test_ct, block_idx)
            
            if test_block == target_block:
                known_flag += c1.encode() + c2.encode()
                print(f"[+] Found pair: {c1}{c2} | Flag: {known_flag.decode()}")
                found_pair = True
                break
        if found_pair:
            break
            
    if not found_pair:
        print("[-] Failed to find pair. Check charset or logic.")
        break

print(f"\n[SUCCESS] Final Flag: {known_flag.decode()}")
r.close()
```
